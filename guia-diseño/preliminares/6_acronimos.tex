\section*{Abreviaturas y acrónimos}
\addcontentsline{toc}{section}{Abreviaturas y acrónimos}
\label{sec::acronyns}

\vspace{10 mm}
\begin{description}[align=right,labelwidth=2cm] 
    \item [3DaM] 3D as Master (3D como Patrón)
    \item [ATA] Air Transport Association of America (Asociación de Transporte Aéreo de América). Generalmente se refiere a la clasificación ``ATA Chapters'' del AMM
    \item [APU] Auxiliary Power Unit (Unidad de Potencia Auxiliar)
    \item [AMM] Aircraft Maintenance Manual (Manual de Mantenimiento del Avión)
    \item [BF] Belly Fairing (Carenado Ventral)
    \item [CAD] Computer Aided Design (Diseño Asistido por Ordenador)
    \item [CI] Configuration Item (Objeto de configuración)
    \item [DDS] Design Data Set (Set de Datos de Diseño)
    \item [DMU] Digital Mock-Up (Modelo Digital)
    \item [DO] Design Office (Oficina de Diseño)
    \item [DOA] Desgin Organization Approval (Organización de Diseño Aprobada)
    \item [DQN] Design Query Note (Notas de Consulta al Diseño)
    \item [DT] Desviación Típica
    \item [ECN] Engineering Change Note (Nota de Cambios de Ingeniería)
    \item [EPIs] Equipos de Protección Individual
    \item [FAL] Final Assembly Line (Línea de Ensamblaje Final)
    \item [FIC] First Implementation Check (Verificación de la primera instalación)
    \item [HTP] Horizontal Tail Plane (Estabilizador Horizontal de Cola)
    \item [LO] Linked Object (Objeto de Anclaje)
    \item [LR] Long Range (Largo Alcance, se refiere al Airbus A330)
    \item [Med.] Media
    \item [MSN] Manufacturer Serial Number (Número de Serie del Fabricante)
    \item [PDM] Product Data Management (Administración de Datos de Producto)
    \item [POA] Production Organization Approval (Organización de Producción Aprobada)
    \item [Pos.] Posición
    \item [RA] Realidad Aumentada
    \item [RFE] Rear Fuselage Empennage (Empenaje posterior del fuselaje)
    \item [RM] Realidad Mixta
    \item [RV] Realidad Virtual
    \item [RX] Realidad Extendida
    \item [SA] Single Aisle (Pasillo Único, se refiere al Airbus A320)
    \item [SOI] Standard Operating Instruction (Instrucciones de Operación Estándar)
    \item [UPM] Universidad Politécnica de Madrid
    \item [VTP] Vertical Tail Plane (Estabilizador Vertical de Cola)
    \item [WB] Wide Body (Fuselaje Ancho, se refiere al Airbus A350)
\end{description}
